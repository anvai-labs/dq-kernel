==== Circuit Breaker Pattern

**Enhancement**: Prevent cascade failures in distributed environments

[source,python]

from enum import Enum
import time
from typing import Callable, Any

class CircuitState(Enum):
    CLOSED = "closed"
    OPEN = "open"
    HALF_OPEN = "half_open"

class CircuitBreaker:
    def __init__(self, failure_threshold: int = 5, timeout: int = 60):
        self.failure_threshold = failure_threshold
        self.timeout = timeout
        self.failure_count = 0
        self.last_failure_time = None
        self.state = CircuitState.CLOSED
    
    def call(self, func: Callable, *args, **kwargs) -> Any:
        if self.state == CircuitState.OPEN:
            if time.time() - self.last_failure_time > self.timeout:
                self.state = CircuitState.HALF_OPEN
            else:
                raise CircuitBreakerOpenError("Circuit breaker is open")
        
        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise

    def _on_success(self):
        self.failure_count = 0
        self.state = CircuitState.CLOSED
    
    def _on_failure(self):
        self.failure_count += 1
        self.last_failure_time = time.time()
        
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN

# Usage in engine loading
engine_circuit_breaker = CircuitBreaker(failure_threshold=3, timeout=120)

def load_engine_with_circuit_breaker(engine_name: str, config: Dict):
    return engine_circuit_breaker.call(
        EngineLoader().load_engine, 
        engine_name, 
        config
    )

from functools import wraps
import hashlib
import pickle

class DQCacheManager:
    def __init__(self):
        self._memory_cache = {}
        self._disk_cache_path = "/tmp/dq_cache"
        self._spark_cache = {}
    
    def cache_key(self, *args, **kwargs) -> str:
        """Generate cache key from function arguments"""
        key_data = pickle.dumps((args, sorted(kwargs.items())))
        return hashlib.md5(key_data).hexdigest()
    
    def dataframe_cache(self, cache_level: str = "memory"):
        """Decorator for caching DataFrame operations"""
        def decorator(func):
            @wraps(func)
            def wrapper(*args, **kwargs):
                cache_key = self.cache_key(*args, **kwargs)
                
                # Check memory cache first
                if cache_key in self._memory_cache:
                    return self._memory_cache[cache_key]
                
                # Execute function
                result = func(*args, **kwargs)
                
                # Cache result based on level
                if cache_level == "memory":
                    self._memory_cache[cache_key] = result
                elif cache_level == "spark":
                    if hasattr(result, 'cache'):
                        result = result.cache()
                        self._spark_cache[cache_key] = result
                
                return result
            return wrapper
        return decorator

# Usage
cache_manager = DQCacheManager()

@cache_manager.dataframe_cache(cache_level="spark")
def get_reference_data(self, table_name: str, columns: List[str]) -> DataFrame:
    return self._spark.table(table_name).select(*columns)

class QueryOptimizer:
    def __init__(self, spark_session: SparkSession):
        self.spark = spark_session
    
    def optimize_dataframe_for_checks(self, df: DataFrame, checks: List[Dict]) -> DataFrame:
        """Optimize DataFrame based on required checks"""
        
        # Extract required columns
        required_columns = set()
        for check in checks:
            if 'columns' in check:
                required_columns.update(check['columns'])
            if 'group_by' in check:
                required_columns.update(check['group_by'])
        
        # Project only required columns
        if required_columns and required_columns.issubset(set(df.columns)):
            df = df.select(*list(required_columns))
        
        # Apply filters early if possible
        filters = self._extract_filters_from_checks(checks)
        for filter_condition in filters:
            df = df.filter(filter_condition)
        
        # Repartition for better parallelism if needed
        if df.rdd.getNumPartitions() < self.spark.sparkContext.defaultParallelism:
            df = df.repartition(self.spark.sparkContext.defaultParallelism)
        
        return df
    
    def _extract_filters_from_checks(self, checks: List[Dict]) -> List[str]:
        """Extract filter conditions that can be pushed down"""
        filters = []
        for check in checks:
            # Example: Extract range filters from compliance checks
            if check.get('constraint') == 'satisfies':
                expression = check.get('expression', '')
                if 'BETWEEN' in expression.upper():
                    filters.append(expression)
        return filters

        self.config = config

    def create_streaming_dq_pipeline(self, input_stream: DataFrame) -> DataFrame:
        """Create streaming DQ pipeline with continuous monitoring"""
        
        def process_micro_batch(batch_df: DataFrame, batch_id: int):
            logger.info("Processing micro-batch", batch_id=batch_id, 
                       record_count=batch_df.count())
            
            # Apply DQ checks to micro-batch
            dq_framework = DQFramework(self.spark, self.config, batch_df)
            results = dq_framework.run()
            
            # Store results with batch metadata
            enriched_results = [
                {**result, "batch_id": batch_id, "processing_time": datetime.utcnow()}
                for result in results
            ]
            
            # Write to results stream
            self._write_streaming_results(enriched_results)
            
            # Trigger alerts for failures
            self._check_and_alert(enriched_results)
        
        # Apply DQ processing to each micro-batch
        query = input_stream.writeStream \
            .foreachBatch(process_micro_batch) \
            .option("checkpointLocation", self.checkpoint_path) \
            .trigger(processingTime='30 seconds') \
            .start()
        
        return query
    
    def _write_streaming_results(self, results: List[Dict]):
        """Write results to streaming sink"""
        results_df = self.spark.createDataFrame(results)
        
        results_df.write \
            .format("delta") \
            .mode("append") \
            .option("path", self.config.streaming.results_path) \
            .save()
    
    def _check_and_alert(self, results: List[Dict]):
        """Check for failures and trigger alerts"""
        failed_checks = [r for r in results if not r.get('success', True)]
        
        if failed_checks:
            alert_manager.send_alert(
                severity="WARNING" if len(failed_checks) < 3 else "CRITICAL",
                message=f"{len(failed_checks)} data quality checks failed",
                details=failed_checks

from sklearn.ensemble import IsolationForest
import numpy as np

class IntelligentAlertManager:
    def __init__(self):
        self.anomaly_detector = IsolationForest(contamination=0.1)
        self.baseline_metrics = {}
        self.alert_channels = []
    
    def train_baseline(self, historical_results: List[Dict]):
        """Train anomaly detection model on historical DQ metrics"""
        
        # Extract numerical features
        features = []
        for result in historical_results:
            feature_vector = [
                result.get('execution_time_ms', 0),
                result.get('record_count', 0),
                1 if result.get('success', True) else 0,
                result.get('metric_value', 0)
            ]
            features.append(feature_vector)
        
        # Train anomaly detector
        if features:
            self.anomaly_detector.fit(np.array(features))
    
    def evaluate_results(self, current_results: List[Dict]) -> List[Dict]:
        """Evaluate current results for anomalies"""
        
        alerts = []
        
        for result in current_results:
            # Rule-based alerts
            if not result.get('success', True):
                alerts.append({
                    'type': 'FAILURE',
                    'severity': 'HIGH',
                    'message': f"Check {result['check']} failed",
                    'details': result
                })
            
            # Anomaly-based alerts
            feature_vector = np.array([[
                result.get('execution_time_ms', 0),
                result.get('record_count', 0),
                1 if result.get('success', True) else 0,
                result.get('metric_value', 0)
            ]])
            
            if hasattr(self.anomaly_detector, 'decision_function'):
                anomaly_score = self.anomaly_detector.decision_function(feature_vector)[0]
                
                if anomaly_score < -0.5:  # Threshold for anomaly
                    alerts.append({
                        'type': 'ANOMALY',
                        'severity': 'MEDIUM',
                        'message': f"Anomalous behavior detected in {result['check']}",
                        'anomaly_score': anomaly_score,
                        'details': result
                    })
        
        return alerts
    
    def send_alerts(self, alerts: List[Dict]):
        """Send alerts through configured channels"""
        for alert in alerts:
            for channel in self.alert_channels:
                channel.send(alert)

from sklearn.ensemble import IsolationForest
from sklearn.preprocessing import StandardScaler
import pandas as pd

class MLDataQualityDetector:
    def __init__(self):
        self.models = {}
        self.scalers = {}
        self.feature_extractors = {}
    
    def train_anomaly_detector(self, table_name: str, historical_data: DataFrame):
        """Train ML model for automatic anomaly detection"""
        
        # Extract statistical features
        features_df = self._extract_statistical_features(historical_data)
        
        # Normalize features
        scaler = StandardScaler()
        features_normalized = scaler.fit_transform(features_df)
        
        # Train isolation forest
        model = IsolationForest(
            contamination=0.05,  # 5% expected anomalies
            random_state=42
        )
        model.fit(features_normalized)
        
        # Store model and scaler
        self.models[table_name] = model
        self.scalers[table_name] = scaler
        
        return model
    
    def detect_anomalies(self, table_name: str, current_data: DataFrame) -> List[Dict]:
        """Detect anomalies in current data using trained model"""
        
        if table_name not in self.models:
            raise ValueError(f"No trained model for table: {table_name}")
        
        # Extract features from current data
        features_df = self._extract_statistical_features(current_data)
        features_normalized = self.scalers[table_name].transform(features_df)
        
        # Predict anomalies
        anomaly_scores = self.models[table_name].decision_function(features_normalized)
        predictions = self.models[table_name].predict(features_normalized)
        
        # Generate anomaly reports
        anomalies = []
        for i, (score, is_anomaly) in enumerate(zip(anomaly_scores, predictions)):
            if is_anomaly == -1:  # Anomaly detected
                anomalies.append({
                    'type': 'ML_ANOMALY',
                    'severity': 'MEDIUM' if score > -0.3 else 'HIGH',
                    'column': features_df.index[i],
                    'anomaly_score': score,
                    'description': f'Statistical anomaly detected in column {features_df.index[i]}',
                    'recommendations': self._generate_recommendations(features_df.index[i], score)
                })
        
        return anomalies
    
    def _extract_statistical_features(self, df: DataFrame) -> pd.DataFrame:
        """Extract statistical features for ML analysis"""
        
        numeric_columns = df.select_dtypes(include=[np.number]).columns
        features = {}
        
        for col in numeric_columns:
            col_data = df[col].dropna()
            if len(col_data) > 0:
                features[f'{col}_mean'] = col_data.mean()
                features[f'{col}_std'] = col_data.std()
                features[f'{col}_skew'] = col_data.skew()
                features[f'{col}_kurtosis'] = col_data.kurtosis()
                features[f'{col}_null_rate'] = df[col].isnull().mean()
                features[f'{col}_unique_rate'] = df[col].nunique() / len(df)
        
        return pd.DataFrame([features])
    
    def _generate_recommendations(self, column: str, anomaly_score: float) -> List[str]:
        """Generate recommendations for addressing anomalies"""
        
        recommendations = []
        
        if 'null_rate' in column:
            recommendations.append("Investigate data source for missing value issues")
            recommendations.append("Consider implementing data imputation strategies")
        
        elif 'mean' in column or 'std' in column:
            recommendations.append("Check for data processing pipeline changes")
            recommendations.append("Validate data source for unusual value distributions")
        
        elif 'unique_rate' in column:
            recommendations.append("Investigate potential data deduplication issues")
            recommendations.append("Check for data key generation problems")
        
        if anomaly_score < -0.5:
            recommendations.append("URGENT: This is a severe anomaly requiring immediate investigation")
        
        return recommendations

class AutoDQRuleGenerator:
    def __init__(self):

                'min': threshold,
                'level': 'Error'
            }
        }
    
    def generate_rules_from_profiling(self, df: DataFrame, table_name: str) -> List[Dict]:
        """Automatically generate DQ rules based on data profiling"""

        # Profile the data
        profile = self._profile_dataframe(df)
        
        for column, stats in profile.items():
            # Generate completeness rules
            if stats['null_rate'] < 0.05:  # Less than 5% nulls historically

                auto_rules.append(rule)
            
            # Generate uniqueness rules for potential key columns

                rule = self.rule_templates['uniqueness'](column, 0.98)
                auto_rules.append(rule)
            
            # Generate range rules for numeric columns
            if stats['data_type'] in ['int', 'float']:
                if stats['min'] >= 0:  # Non-negative values
                    auto_rules.append({
                        'constraint_name': f'{column}_non_negative_auto',
                        'constraint': 'satisfies',
                        'expression': f'{column} >= 0',
                        'min': 0.99,
                        'level': 'Warning'
                    })
        
        return auto_rules
    
    def _profile_dataframe(self, df: DataFrame) -> Dict:
        """Profile DataFrame to extract statistical characteristics"""
        
        profile = {}
        
        for column in df.columns:
            col_stats = {
                'data_type': str(df.schema[column].dataType),
                'null_rate': df.filter(F.col(column).isNull()).count() / df.count(),
                'unique_rate': df.select(column).distinct().count() / df.count()
            }
            
            # Numeric statistics
            if isinstance(df.schema[column].dataType, (IntegerType, DoubleType, FloatType)):
                stats_row = df.agg(
                    F.min(column).alias('min'),
                    F.max(column).alias('max'),
                    F.mean(column).alias('mean'),
                    F.stddev(column).alias('std')
                ).collect()[0]
                
                col_stats.update({
                    'min': stats_row['min'],
                    'max': stats_row['max'],
                    'mean': stats_row['mean'],
                    'std': stats_row['std']
                })
            
            profile[column] = col_stats
        
        return profile
