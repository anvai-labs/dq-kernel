def _check_rate_of_change_optimized(self, dq_dimension, constraint, dataframe, 
                                   columns=None, group_by=None, sort_by=None, 
                                   min=None, max=None, level="Warning"):
    """Optimized rate of change check using distributed processing"""
    
    window = Window.partitionBy(*group_by).orderBy(F.col(sort_by))
    
    # Process all columns in single pass
    for column in columns:
        # Calculate rate of change using window functions
        dataframe = dataframe.withColumn(
            f"{column}_prev", 
            F.lag(F.col(column), 1).over(window)
        ).withColumn(
            f"{column}_change_pct",
            F.when(F.col(f"{column}_prev") != 0,
                   F.abs((F.col(column) - F.col(f"{column}_prev")) / 
                         F.col(f"{column}_prev") * 100))
            .otherwise(0)
        )
    
    # Apply thresholds and aggregate results using Spark operations
    violation_checks = []
    for column in columns:
        violation_count = dataframe.filter(
            F.col(f"{column}_change_pct") > max
        ).count()
        
        total_count = dataframe.filter(
            F.col(f"{column}_prev").isNotNull()
        ).count()
        
        success = violation_count == 0 if max else True
        
        violation_checks.append({
            "constraint": constraint,
            "column": column,
            "violations": violation_count,
            "total_records": total_count,
            "success": success,
            "violation_rate": violation_count / total_count if total_count > 0 else 0
        })
    
    return self._format_check_results(violation_checks, dq_dimension, level)

class ReferenceDataManager:
    def __init__(self, spark_session: SparkSession):
        self.spark = spark_session
        self._cache = {}
        self._broadcast_vars = {}
    
    def get_reference_data(self, table_name: str, column_name: str, 
                          cache_strategy: str = "broadcast") -> Set[str]:
        """Get reference data with intelligent caching"""
        
        cache_key = f"{table_name}.{column_name}"
        
        if cache_key in self._cache:
            return self._cache[cache_key]
        
        # Query reference data once
        ref_df = self.spark.sql(f"SELECT DISTINCT {column_name} FROM {table_name}")
        
        # Choose caching strategy based on data size
        ref_count = ref_df.count()
        
        if cache_strategy == "broadcast" and ref_count < 10000:
            # Small reference data - broadcast to all nodes
            ref_list = [row[0] for row in ref_df.collect()]
            ref_set = set(ref_list)
            self._broadcast_vars[cache_key] = self.spark.sparkContext.broadcast(ref_set)
            self._cache[cache_key] = ref_set
        else:
            # Large reference data - cache DataFrame
            ref_df = ref_df.cache()
            ref_list = [row[0] for row in ref_df.collect()]
            self._cache[cache_key] = set(ref_list)
        
        return self._cache[cache_key]
    
    def validate_columns_against_reference(self, dataframe: DataFrame, 
                                         ref_table: str, ref_column: str,
                                         ignore_columns: List[str] = None) -> Dict:
        """Validate DataFrame columns against reference data efficiently"""
        
        ref_data = self.get_reference_data(ref_table, ref_column)
        
        columns_to_check = [col for col in dataframe.columns 
                           if col not in (ignore_columns or [])]
        
        validation_results = {}
        
        for column in columns_to_check:
            # Use broadcast variable if available
            cache_key = f"{ref_table}.{ref_column}"
            if cache_key in self._broadcast_vars:
                # Distributed validation using broadcast
                broadcast_ref = self._broadcast_vars[cache_key]
                
                def is_valid_column(col_name):
                    return col_name in broadcast_ref.value
                
                is_valid_udf = F.udf(is_valid_column, BooleanType())
                
                # This approach validates existence without bringing data to driver
                validation_count = dataframe.select(
                    F.lit(column).alias("column_name")
                ).withColumn(
                    "is_valid", is_valid_udf(F.lit(column))
                ).agg(
                    F.sum(F.col("is_valid").cast("int")).alias("valid_count")
                ).collect()[0]["valid_count"]
                
                validation_results[column] = {
                    "column": column,
                    "is_valid": validation_count > 0,
                    "found_in_reference": column in ref_data
                }
            else:
                # Fallback to simple check
                validation_results[column] = {
                    "column": column,
                    "is_valid": column in ref_data,
                    "found_in_reference": column in ref_data
                }
        
        return validation_results

class IntelligentCacheManager:
    def __init__(self, spark_session: SparkSession):
        self.spark = spark_session
        self._cached_dataframes = {}
        self._cache_stats = {}
        self._memory_threshold = 0.8  # 80% of available memory
    
    def smart_cache(self, dataframe: DataFrame, cache_key: str, 
                   access_pattern: str = "multiple") -> DataFrame:
        """Intelligently cache DataFrame based on usage patterns"""
        
        if cache_key in self._cached_dataframes:
            self._update_access_stats(cache_key)
            return self._cached_dataframes[cache_key]
        
        # Estimate DataFrame size
        estimated_size = self._estimate_dataframe_size(dataframe)
        current_memory_usage = self._get_current_memory_usage()
        
        # Decide caching strategy
        if estimated_size < 100 * 1024 * 1024:  # < 100MB
            storage_level = StorageLevel.MEMORY_ONLY
        elif estimated_size < 1024 * 1024 * 1024:  # < 1GB
            storage_level = StorageLevel.MEMORY_AND_DISK
        else:
            storage_level = StorageLevel.DISK_ONLY
        
        # Apply caching if memory allows
        if current_memory_usage < self._memory_threshold:
            cached_df = dataframe.persist(storage_level)
            self._cached_dataframes[cache_key] = cached_df
            self._cache_stats[cache_key] = {
                "access_count": 1,
                "last_access": time.time(),
                "size_estimate": estimated_size,
                "storage_level": storage_level
            }
            return cached_df
        else:
            # Memory pressure - evict least recently used
            self._evict_lru_cache()
            return self.smart_cache(dataframe, cache_key, access_pattern)
    
    def _estimate_dataframe_size(self, dataframe: DataFrame) -> int:
        """Estimate DataFrame memory footprint"""
        
        # Sample-based size estimation
        sample_fraction = min(0.01, 1000.0 / dataframe.count())  # 1% or 1000 rows
        sample_df = dataframe.sample(withReplacement=False, fraction=sample_fraction)
        
        # Calculate size based on data types and row count
        total_rows = dataframe.count()
        sample_rows = sample_df.count()
        
        if sample_rows == 0:
            return 0
        
        # Estimate based on schema and sample
        size_per_row = 0
        for field in dataframe.schema.fields:
            if isinstance(field.dataType, StringType):
                size_per_row += 50  # Average string size
            elif isinstance(field.dataType, IntegerType):
                size_per_row += 4
            elif isinstance(field.dataType, DoubleType):
                size_per_row += 8
            else:
                size_per_row += 20  # Default estimate
        
        estimated_size = total_rows * size_per_row
        return estimated_size
    
    def _evict_lru_cache(self):
        """Evict least recently used cached DataFrames"""
        
        if not self._cache_stats:
            return
        
        # Find least recently used
        lru_key = min(self._cache_stats.keys(), 
                     key=lambda k: self._cache_stats[k]["last_access"])
        
        # Unpersist and remove
        if lru_key in self._cached_dataframes:
            self._cached_dataframes[lru_key].unpersist()
            del self._cached_dataframes[lru_key]
            del self._cache_stats[lru_key]

class PartitionOptimizer:
    def __init__(self, spark_session: SparkSession):
        self.spark = spark_session
        self.default_parallelism = spark_session.sparkContext.defaultParallelism
    
    def optimize_partitioning(self, dataframe: DataFrame, 
                             operation_type: str = "general") -> DataFrame:
        """Optimize DataFrame partitioning based on operation type"""
        
        current_partitions = dataframe.rdd.getNumPartitions()
        optimal_partitions = self._calculate_optimal_partitions(
            dataframe, operation_type
        )
        
        if abs(current_partitions - optimal_partitions) > 2:
            if operation_type == "aggregation":
                # Partition by key columns for better aggregation performance
                return dataframe.repartition(optimal_partitions)
            elif operation_type == "join":
                # Coalesce to reduce shuffle overhead
                return dataframe.coalesce(optimal_partitions)
            else:
                # General optimization
                if current_partitions < optimal_partitions:
                    return dataframe.repartition(optimal_partitions)
                else:
                    return dataframe.coalesce(optimal_partitions)
        
        return dataframe
    
    def _calculate_optimal_partitions(self, dataframe: DataFrame, 
                                    operation_type: str) -> int:
        """Calculate optimal number of partitions"""
        
        # Estimate data size
        row_count = dataframe.count()
        
        if row_count < 10000:
            return 1
        elif row_count < 100000:
            return min(4, self.default_parallelism)
        elif row_count < 1000000:
            return min(8, self.default_parallelism)
        else:
            # For large datasets, aim for 128MB per partition
            estimated_size_mb = row_count * len(dataframe.columns) * 8 / (1024 * 1024)
            optimal_partitions = max(1, int(estimated_size_mb / 128))
            return min(optimal_partitions, self.default_parallelism * 2)

class QueryOptimizer:
    def __init__(self, spark_session: SparkSession):
        self.spark = spark_session
    
    def optimize_for_dq_checks(self, dataframe: DataFrame, 
                               dq_checks: List[Dict]) -> DataFrame:
        """Optimize DataFrame based on DQ check requirements"""
        
        # Extract optimization hints from checks
        optimization_plan = self._analyze_checks(dq_checks)
        
        # Apply column pruning
        if optimization_plan["required_columns"]:
            dataframe = dataframe.select(*optimization_plan["required_columns"])
        
        # Apply early filters
        for filter_condition in optimization_plan["filters"]:
            dataframe = dataframe.filter(filter_condition)
        
        # Apply sampling for statistical checks
        if optimization_plan["can_sample"]:
            sample_fraction = min(0.1, 100000 / dataframe.count())
            if sample_fraction < 1.0:
                dataframe = dataframe.sample(withReplacement=False, fraction=sample_fraction)
        
        return dataframe
    
    def _analyze_checks(self, dq_checks: List[Dict]) -> Dict:
        """Analyze DQ checks to extract optimization opportunities"""
        
        required_columns = set()
        filters = []
        can_sample = True
        
        for check in dq_checks:
            # Extract required columns
            if check.get("columns"):
                required_columns.update(check["columns"])
            if check.get("group_by"):
                required_columns.update(check["group_by"])
            
            # Extract filter conditions
            if check.get("constraint") == "satisfies":
                expression = check.get("expression", "")
                # Simple filter extraction
                if ">" in expression or "<" in expression:
                    filters.append(expression)
            
            # Check if sampling is appropriate
            if check.get("constraint") in ["hasUniqueness", "RateOfChange"]:
                can_sample = False  # These require full dataset
        
        return {
            "required_columns": list(required_columns) if required_columns else None,
            "filters": filters,
            "can_sample": can_sample
        }

class SQLOptimizedEngine:
    def __init__(self, spark_session: SparkSession):

    def execute_optimized_completeness_check(self, dataframe: DataFrame, 
                                           columns: List[str]) -> Dict:
        """Optimized completeness check using pure SQL"""
        
        # Register DataFrame as temporary view
        view_name = f"dq_temp_view_{int(time.time())}"
        dataframe.createOrReplaceTempView(view_name)
        
        # Build optimized SQL query
        total_count_sql = f"SELECT COUNT(*) as total FROM {view_name}"
        
        null_checks = []
        for col in columns:
            null_checks.append(f"SUM(CASE WHEN {col} IS NULL THEN 1 ELSE 0 END) as {col}_nulls")
        
        null_count_sql = f"""
        SELECT {', '.join(null_checks)}
        FROM {view_name}
        """
        
        # Execute queries
        total_count = self.spark.sql(total_count_sql).collect()[0]["total"]
        null_counts = self.spark.sql(null_count_sql).collect()[0]
        
        # Calculate completeness
        results = {}
        for col in columns:
            null_count = null_counts[f"{col}_nulls"]
            completeness = (total_count - null_count) / total_count if total_count > 0 else 0
            results[col] = {
                "constraint": "hasCompleteness",
                "column": col,
                "value": completeness,
                "total_records": total_count,
                "null_records": null_count
            }
        
        # Cleanup
        self.spark.catalog.dropTempView(view_name)
        
        return results

from concurrent.futures import ThreadPoolExecutor, as_completed
import threading
from typing import Callable, List, Dict, Any

class ParallelDQExecutor:
    def __init__(self, max_workers: int = None):
        self.max_workers = max_workers or min(32, (os.cpu_count() or 1) + 4)
        self._thread_local = threading.local()
    
    def execute_checks_parallel(self, dataframe: DataFrame, 
                               check_configs: List[Dict],
                               engine_factory: Callable) -> List[Dict]:
        """Execute independent DQ checks in parallel"""
        
        # Group checks by independence
        independent_groups = self._group_independent_checks(check_configs)
        
        all_results = []
        
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            # Submit independent check groups
            futures = {}
            
            for group_id, checks in independent_groups.items():
                future = executor.submit(
                    self._execute_check_group, 
                    dataframe, checks, engine_factory
                )
                futures[future] = group_id
            
            # Collect results as they complete
            for future in as_completed(futures):
                group_id = futures[future]
                try:
                    group_results = future.result()
                    all_results.extend(group_results)
                except Exception as e:
                    logger.error(f"Check group {group_id} failed: {e}")
                    # Continue with other groups
        
        return all_results
    
    def _group_independent_checks(self, check_configs: List[Dict]) -> Dict[str, List[Dict]]:
        """Group checks that can be executed independently"""
        
        # Simple grouping by engine type
        groups = {}
        
        for i, check in enumerate(check_configs):
            engine_type = check.get("engine", "unknown")
            group_key = f"{engine_type}_{i // 3}"  # Group in batches of 3
            
            if group_key not in groups:
                groups[group_key] = []
            groups[group_key].append(check)
        
        return groups
    
    def _execute_check_group(self, dataframe: DataFrame, 
                           checks: List[Dict], 
                           engine_factory: Callable) -> List[Dict]:
        """Execute a group of checks"""
        
        results = []
        
        for check in checks:
            try:
                engine = engine_factory(check)
                check_result = engine.apply(dataframe)
                results.extend(check_result)
            except Exception as e:
                logger.error(f"Check {check.get('constraint_name', 'unknown')} failed: {e}")
                # Add error result
                results.append({
                    "check": check.get("constraint_name", "unknown"),
                    "success": False,
                    "error": str(e),
                    "details": {"error_type": type(e).__name__}
                })
        
        return results

import time

from dataclasses import dataclass

@dataclass
class PerformanceMetrics:
    execution_time: float
    memory_usage_mb: float
    cpu_usage_percent: float
    disk_io_mb: float

class PerformanceProfiler:
    def __init__(self, spark_session: SparkSession):
        self.spark = spark_session
        self._metrics_history = []
    
    @contextmanager
    def profile_execution(self, operation_name: str):
        """Context manager for profiling DQ operations"""
        
        # Capture initial state
        start_time = time.time()
        initial_memory = psutil.virtual_memory().used / (1024 * 1024)
        initial_cpu = psutil.cpu_percent()
        
        # Spark metrics before
        spark_metrics_before = self._get_spark_metrics()
        
        try:
            yield
        finally:
            # Capture final state
            end_time = time.time()
            final_memory = psutil.virtual_memory().used / (1024 * 1024)
            final_cpu = psutil.cpu_percent()
            
            # Spark metrics after
            spark_metrics_after = self._get_spark_metrics()
            
            # Calculate metrics
            metrics = PerformanceMetrics(
                execution_time=end_time - start_time,
                memory_usage_mb=final_memory - initial_memory,
                cpu_usage_percent=(final_cpu + initial_cpu) / 2,
                disk_io_mb=0,  # Would need more sophisticated tracking
                network_io_mb=0,  # Would need more sophisticated tracking
                spark_metrics=self._calculate_spark_delta(
                    spark_metrics_before, spark_metrics_after
                )
            )
            
            # Store metrics
            self._metrics_history.append({
                "operation": operation_name,
                "timestamp": end_time,
                "metrics": metrics
            })
            
            # Log performance
            logger.info(f"Performance metrics for {operation_name}",
                       execution_time=metrics.execution_time,
                       memory_usage_mb=metrics.memory_usage_mb,
                       cpu_usage=metrics.cpu_usage_percent)
    
    def _get_spark_metrics(self) -> Dict[str, Any]:
        """Get current Spark metrics"""
        
        try:
            # Access Spark UI metrics programmatically
            sc = self.spark.sparkContext
            
            return {
                "tasks_completed": sc.statusTracker().getExecutorInfos(),
                "active_jobs": len(sc.statusTracker().getActiveJobIds()),
                "active_stages": len(sc.statusTracker().getActiveStageIds())
            }
        except Exception:
            return {}
    
    def _calculate_spark_delta(self, before: Dict, after: Dict) -> Dict:
        """Calculate difference in Spark metrics"""
        
        delta = {}
        
        # Simple delta calculation
        for key in before:
            if key in after and isinstance(before[key], (int, float)):
                delta[f"{key}_delta"] = after[key] - before[key]
        
        return delta
    
    def get_performance_report(self, operation_filter: str = None) -> Dict:
        """Generate performance report"""
        
        filtered_metrics = self._metrics_history
        if operation_filter:
            filtered_metrics = [
                m for m in self._metrics_history 
                if operation_filter in m["operation"]
            ]
        
        if not filtered_metrics:
            return {"message": "No metrics available"}
        
        # Calculate statistics
        execution_times = [m["metrics"].execution_time for m in filtered_metrics]
        memory_usage = [m["metrics"].memory_usage_mb for m in filtered_metrics]
        
        return {
            "total_operations": len(filtered_metrics),
            "avg_execution_time": sum(execution_times) / len(execution_times),
            "max_execution_time": max(execution_times),
            "min_execution_time": min(execution_times),
            "avg_memory_usage": sum(memory_usage) / len(memory_usage),
            "max_memory_usage": max(memory_usage),
            "recent_operations": filtered_metrics[-10:]  # Last 10 operations
        }

            results = []
